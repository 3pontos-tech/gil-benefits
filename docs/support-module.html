ticket_destinations

Uma linha por destino por chamado. Permite múltiplos destinos futuros sem alterar o schema principal.

Schema::create('ticket_destinations', function (Blueprint $table) {
    $table->ulid('id')->primary();
    $table->foreignUlid('support_ticket_id')->constrained()->cascadeOnDelete();
    $table->string('type');     // TicketDestinationTypeEnum: monday | email
    $table->string('channel');  // TicketDestinationChannelEnum
    $table->string('reference_id')->nullable(); // Resend email ID ou Monday item ID
    $table->string('status')->default('pending'); // pending | sent | failed
    $table->timestamps();
});migration
Tracking de e-mail — o reference_id armazena o ID do Resend retornado pelo basementdevs/filament-better-mails. O histórico de eventos de entrega (delivered, bounced, etc.) vive nas tabelas do better-mails, não aqui.

Protocolo #

Formato SUP-{ano}-{seq} — sequencial por ano com reset anual, zero-pad de 4 dígitos. A PK do ticket continua sendo ULID; o protocolo é apenas para exibição ao usuário.

class ProtocolGenerator
{
    public function generate(): string
    {
        $year = now()->year;

        $last = SupportTicket::query()
            ->whereYear('created_at', $year)
            ->orderByDesc('created_at')
            ->value('protocol');

        $seq = $last ? ((int) substr($last, -4)) + 1 : 1;

        return sprintf('SUP-%d-%04d', $year, $seq);
    }
}ProtocolGenerator.php
Atenção em produção: com concorrência alta, dois tickets criados simultaneamente podem gerar o mesmo protocolo. Mitigar com DB::transaction() + lock pessimista na leitura do último protocolo, ou com uma sequence dedicada no banco.

CreateSupportTicketAction #

class CreateSupportTicketAction
{
    public function __construct(
        private readonly ProtocolGenerator $protocols,
    ) {}

    public function execute(CreateSupportTicketDTO $dto): SupportTicket
    {
        $ticket = SupportTicket::create([
            'protocol'             => $this->protocols->generate(),
            'user_id'              => $dto->userId,
            'company_id'           => $dto->companyId,
            'visitor_name'         => $dto->visitorName,
            'visitor_email'        => $dto->visitorEmail,
            'visitor_company_name' => $dto->visitorCompanyName,
            'category'             => $dto->category,
            'subject'              => $dto->subject,
            'description'          => $dto->description,
            'status'               => SupportTicketStatusEnum::Pending,
            'url'                  => $dto->url,
            'browser'              => $dto->browser,
            'device'               => $dto->device,
            'environment'          => $dto->environment,
        ]);

        if ($dto->attachment) {
            $ticket->addMedia($dto->attachment)
                   ->toMediaCollection('attachments');
        }

        DispatchSupportTicketJob::dispatch($ticket);

        return $ticket;
    }
}CreateSupportTicketAction.php

DispatchSupportTicketJob #

Executa o roteamento de forma assíncrona. 3 tentativas com backoff de 30s. Em caso de falha total, atualiza o status do ticket para Failed.

class DispatchSupportTicketJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $tries   = 3;
    public int $backoff = 30;

    public function __construct(
        public readonly SupportTicket $ticket
    ) {}

    public function handle(TicketRouterService $router): void
    {
        $router->dispatch($this->ticket);
    }

    public function failed(Throwable $e): void
    {
        $this->ticket->update([
            'status' => SupportTicketStatusEnum::Failed,
        ]);
    }
}DispatchSupportTicketJob.php

TicketRouterService #

class TicketRouterService
{
    public function dispatch(SupportTicket $ticket): void
    {
        $channel = $ticket->category->getDestinationChannel();
        $type    = $channel->getDestinationType();

        $destination = TicketDestination::create([
            'support_ticket_id' => $ticket->id,
            'type'              => $type,
            'channel'           => $channel,
            'status'            => 'pending',
        ]);

        match ($type) {
            TicketDestinationTypeEnum::Email  => $this->routeEmail($ticket, $destination),
            TicketDestinationTypeEnum::Monday => $this->routeMonday($ticket, $destination),
        };

        $ticket->update(['status' => SupportTicketStatusEnum::Dispatched]);
    }

    private function routeEmail(SupportTicket $ticket, TicketDestination $dest): void
    {
        // e-mail para a área interna
        $sent = Mail::to($dest->channel->getRecipientEmail())
            ->send(new SupportTicketMail($ticket));

        $dest->update([
            'status'       => 'sent',
            'reference_id' => $sent->getMessageId(), // ID Resend → better-mails
        ]);

        // confirmação para o solicitante (autenticado ou visitante)
        $to = $ticket->user?->email ?? $ticket->visitor_email;
        Mail::to($to)->send(new SupportTicketConfirmationMail($ticket));
    }

    private function routeMonday(SupportTicket $ticket, TicketDestination $dest): void
    {
        // fase 2
        throw new \RuntimeException('Monday integration not implemented.');
    }
}TicketRouterService.php

Rate Limiting #

Aplicado apenas na rota pública do formulário de visitante externo (US-03). Usuários autenticados não são limitados.

// AppServiceProvider::boot()
RateLimiter::for('support-public', function (Request $request) {
    return [
        Limit::perHour(3)->by('email:' . $request->input('visitor_email')),
        Limit::perDay(5)->by('email_day:' . $request->input('visitor_email')),
        Limit::perHour(10)->by('ip:' . $request->ip()),
        Limit::perDay(20)->by('ip_day:' . $request->ip()),
    ];
});
